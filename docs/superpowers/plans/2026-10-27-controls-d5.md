```csharp
namespace SharpUI;

/// <summary>浮层条目(每窗栈;F-1)。</summary>
internal sealed class Popup
{
    public Element Root { get; }                 // 浮层根元素(Focusable 树)
    public RectF AnchorRect { get; }             // 锚定矩形(逻辑坐标;Dialog=全屏居中忽略)
    public PopupKind Kind { get; }               // Dialog / Menu / Tooltip
    public bool DismissOnOutsideClick { get; init; } = true;
    public Action? OnClosed { get; init; }       // 关闭回调(焦点归还等收尾)
}
internal enum PopupKind { Dialog, Menu, Tooltip }

/// <summary>每窗浮层栈(静态按窗服务定位,同 App.Atlas 模式;App.Run 每帧接线)。</summary>
public static class PopupHost
{
    public static void Push(IntPtr sdlWindow, Popup popup);
    public static void Pop(IntPtr sdlWindow);
    public static IReadOnlyList<Popup> StackOf(IntPtr sdlWindow);   // 测试观测
    public static int Count(IntPtr sdlWindow);
    // App.Run 集成(实现者按 App.cs 现状选接线点,报告写明):
    //   绘制:声明树 Draw 后,栈底→栈顶逐层 Draw(后进在上)
    //   命中:Interaction.Update 前将栈顶浮层根注入命中(或 Update 后覆盖 HoveredId——以 Interaction
    //     现有结构最小侵入为准,外部点击(命中落浮层外)且 DismissOnOutsideClick → Pop)
    //   键盘:栈非空时 Esc → Pop 栈顶(Tooltip 除外);Focus 围栏生效
}

// Focus.cs 增(F-2):
public sealed class FocusManager
{
    public void PushScope(string scopeId);        // 围栏入栈:Tab/方向键导航限制在栈顶 scope 元素内
    public void PopScope();                        // 出栈:焦点归还 push 时的 FocusedId(打开者)
    public string? ActiveScopeId { get; }          // 栈顶 scope(空=无围栏)
}
```

- [ ] **Step 1: 写失败测试**(PopupHostTests.cs,模式同 ControlsTests——合成事件+Interaction.Update+FocusManager):
  - Push/Pop/Count/StackOf 后进先出;两窗隔离(不同 sdlWindow 句柄各自栈)。
  - 命中优先:栈有浮层时,合成 Motion 落声明树元素上 → HoveredId 为 null 或浮层元素(非声明树元素);落浮层元素 → 浮层元素。
  - 外部点击:DismissOnOutsideClick=true 时点击浮层外 → Pop+OnClosed;=false 不 Pop;点浮层内不 Pop。
  - Esc:栈非空按 Esc → Pop 栈顶;Tooltip 不响应 Esc。
  - 焦点围栏:PushScope 后 Tab 只在 scope 内循环(焦点不出圈);PopScope 焦点归还 push 时值。
- [ ] **Step 2-5: 红→实现(App.Run/Interaction/Focus 接线按现状最小侵入,报告写明各接线点)→全量绿→提交**

namespace SharpUI;

public sealed class MenuItem(string label, Action onClick)

public static class Ui
{
    /// <summary>在锚定矩形处弹出菜单(命令式:按钮点击回调内调用;锚=触发元素 Frame 底部)。</summary>
    public static void ShowMenu(IntPtr sdlWindow, RectF anchorRect, params object?[] items);
    // 语义:PopupHost.Push(Menu 类,锚定);方向键上下移动高亮(禁用项跳过);Enter 激活+关闭;
    //   Esc/外部点击关闭;焦点围栏(打开时);激活后 Pop+OnClosed 归还;
    //   菜单项视觉:高亮=悬停蓝 0x3B82F6,禁用=灰 0x3A3A42,分隔线=1px 0x2A2A30
}

public static class ElementExtensions
{
    /// <summary>给任意元素附悬停提示(F-1 Tooltip 经 PopupHost;链式)。</summary>
    public static T Tooltip<T>(this T e, string text) where T : Element;

    //   锚=元素 Frame);移出/点击/焦点变化 → Pop;无焦点语义;同元素重复悬停不重复 Push(在栈判重)
}
